import { useState, useCallback, useMemo } from 'react'
import Header from './components/layout/Header'
import Sidebar from './components/layout/Sidebar'
import BusinessGraph from './components/twin/BusinessGraph'
import InsightPanel from './components/intelligence/InsightPanel'
import AskNexus from './components/intelligence/AskNexus'
import { useBusinessGraph } from './hooks/useBusinessGraph'

export default function App() {
  const graph = useBusinessGraph()
  const [query, setQuery] = useState('')
  const [activeScenario, setActiveScenario] = useState(null)

  const handleScenarioSelect = useCallback(
    (scenario) => {
      setQuery(scenario.query)
      setActiveScenario(scenario)
      graph.applyScenarioHighlight(scenario.affectedNodeIds, scenario.focusNodeId)
    },
    [graph]
  )

  const handleClearScenario = useCallback(() => {
    setActiveScenario(null)
    setQuery('')
    graph.clearScenarioHighlight()
    graph.clearSelection()
  }, [graph])

  // Clicking a node directly (focus mode) always supersedes an active
  // scenario preview, so the Intelligence panel and graph never show two
  // contradictory stories at once.
  const handleSelectNode = useCallback(
    (nodeId) => {
      setActiveScenario(null)
      setQuery('')
      graph.selectNode(nodeId)
    },
    [graph]
  )

  // "View dependency" in the Hidden Dependency insight focuses the graph
  // on Product X without going through the scenario/simulation flow.
  const handleViewDependency = useCallback(
    (nodeId) => {
      graph.clearScenarioHighlight()
      graph.selectNode(nodeId)
    },
    [graph]
  )

  // BusinessGraph reads `selectNode` off this object directly, so routing
  // it through the wrapper above keeps App.jsx's scenario state in sync
  // without changing the hook's public shape for other consumers.
  const graphForTwin = useMemo(() => ({ ...graph, selectNode: handleSelectNode }), [graph, handleSelectNode])

  return (
    <div className="app-shell">
      <Header />
      <main className="app-main">
        <Sidebar />
        <BusinessGraph graph={graphForTwin} />
        <InsightPanel
          activeScenario={activeScenario}
          selectedNode={graph.selectedNode}
          onViewDependency={handleViewDependency}
          onClearScenario={handleClearScenario}
        />
      </main>
      <AskNexus
        query={query}
        onQueryChange={setQuery}
        onScenarioSelect={handleScenarioSelect}
        activeScenarioId={activeScenario?.id}
      />
    </div>
  )
}

import { useState } from 'react'
import { Sparkles, ArrowRight } from 'lucide-react'
import ScenarioCards from '../dashboard/ScenarioCards'
import { scenarios } from '../../data/scenarios'

/**
 * Phase 1 Ask NEXUS bar. Scenario chips drive a controlled mock cascade
 * preview (see App.jsx). Free-text input that doesn't match a known
 * scenario is handled honestly — no fake AI response — since natural
 * language interpretation is a Phase 5 capability (see aiService.js).
 */
export default function AskNexus({ query, onQueryChange, onScenarioSelect, activeScenarioId }) {
  const [unmatched, setUnmatched] = useState(false)

  function handleSubmit(e) {
    e.preventDefault()
    const match = scenarios.find((s) => s.query.toLowerCase() === query.trim().toLowerCase())
    if (match) {
      setUnmatched(false)
      onScenarioSelect(match)
    } else if (query.trim().length > 0) {
      setUnmatched(true)
    }
  }

  function handleChipSelect(scenario) {
    setUnmatched(false)
    onScenarioSelect(scenario)
  }

  return (
    <div className="command-zone">
      <ScenarioCards onSelect={handleChipSelect} activeScenarioId={activeScenarioId} />

      <form className="command-bar" onSubmit={handleSubmit}>
        <Sparkles size={17} className="command-bar-icon" />
        <input
          className="command-input"
          placeholder="What would you like to stress-test?"
          value={query}
          onChange={(e) => {
            onQueryChange(e.target.value)
            if (unmatched) setUnmatched(false)
          }}
        />
        <button type="submit" className="command-submit">
          Ask
          <ArrowRight size={14} />
        </button>
      </form>

      {unmatched && (
        <p className="signal-caption" style={{ paddingLeft: 4 }}>
          NEXUS understands guided scenarios in this preview — try one of the suggestions above. Free-form
          reasoning arrives in a later phase.
        </p>

import { useMemo } from 'react'
import { ACCENT } from '../../utils/constants'

// Critical (primary supply→revenue chain) edges get the NEXUS signature
// lime — they're the main artery of the ecosystem. Signal edges (Razorpay)
// get intelligence violet. Backup/standard paths stay neutral graphite.
const EDGE_COLOR = {
  critical: ACCENT.LIME,
  backup: '#5f6861',
  standard: '#5f6861',
  signal: ACCENT.INTELLIGENCE,
}

function hashSign(id) {
  let sum = 0
  for (let i = 0; i < id.length; i++) sum += id.charCodeAt(i)
  return sum % 2 === 0 ? 1 : -1
}

/** Builds an organic quadratic curve — direction alternates per edge (via a
 * deterministic hash of its id) so the graph doesn't read as a mechanical
 * fan of identically-bowed lines. */
function buildCurve(source, target, edgeId) {
  const mx = (source.x + target.x) / 2
  const my = (source.y + target.y) / 2
  const dx = target.x - source.x
  const dy = target.y - source.y
  const dist = Math.sqrt(dx * dx + dy * dy) || 1
  const sign = hashSign(edgeId)
  const offset = Math.min(46, dist * 0.17) * sign
  const cx = mx - (dy / dist) * offset
  const cy = my + (dx / dist) * offset
  return `M ${source.x} ${source.y} Q ${cx} ${cy} ${target.x} ${target.y}`
}

export default function DependencyEdge({
  edge,
  sourceNode,
  targetNode,
  isDimmed,
  isHighlighted,
  particleSeed,
  reducedMotion = false,
  isCascadeActive = false,
  focusDirection, // 'in' | 'out' | undefined — only set during plain node focus (not scenario)
}) {
  const path = useMemo(() => buildCurve(sourceNode, targetNode, edge.id), [sourceNode, targetNode, edge.id])
  const baseColor = EDGE_COLOR[edge.kind] || EDGE_COLOR.standard
  const isDashed = edge.kind === 'backup'
  const pathId = `path-${edge.id}`
  const emphasized = isHighlighted || isCascadeActive
  const isHiddenEdge = Boolean(edge.hidden)

  // Hidden dependencies stay nearly invisible until NEXUS "discovers" them —
  // then they reveal in intelligence violet, distinct from their normal kind color.
  let color = baseColor
  if (isCascadeActive) color = 'var(--status-critical)'
  else if (isHiddenEdge && emphasized) color = ACCENT.INTELLIGENCE
  else if (!isHiddenEdge && emphasized && focusDirection === 'in') color = 'var(--status-warning)'
  else if (!isHiddenEdge && emphasized && focusDirection === 'out') color = ACCENT.LIME

  const restingOpacity = isHiddenEdge ? 0.05 : edge.kind === 'critical' ? 0.5 : 0.32

  return (
    <g>
      <path
        id={pathId}
        className={`dep-edge ${isDimmed ? 'dimmed' : ''} ${emphasized ? 'highlighted' : ''} ${isHiddenEdge ? 'hidden-dep' : ''}`}
        d={path}
        stroke={color}
        strokeWidth={emphasized ? (isHiddenEdge ? 2.2 : 2.6) : edge.kind === 'critical' ? 1.8 : 1.2}
        strokeOpacity={emphasized ? (isHiddenEdge ? 0.85 : 0.9) : restingOpacity}
        strokeDasharray={isDashed ? '4 5' : undefined}
        style={{ transition: 'stroke 0.35s ease, stroke-width 0.35s ease, stroke-opacity 0.35s ease' }}
      />

      {/* Animated flow particle — only on non-dimmed, meaningful paths to avoid visual
          noise, and skipped outright under prefers-reduced-motion (SMIL animateMotion

      {!reducedMotion && !isDimmed && (edge.kind === 'critical' || edge.kind === 'signal' || emphasized) && (

          </animateMotion>

      )}

  )
}

function wrapText(str, maxChars, maxLines) {
  const words = str.split(' ')
  const lines = []
  let current = ''
  for (const w of words) {
    const candidate = current ? `${current} ${w}` : w
    if (candidate.length > maxChars && current) {
      lines.push(current)
      current = w
    } else {
      current = candidate
    }
    if (lines.length === maxLines) break
  }
  if (current && lines.length < maxLines) lines.push(current)
  if (lines.length === maxLines && words.join(' ').length > lines.join(' ').length) {
    lines[maxLines - 1] = lines[maxLines - 1].replace(/\s*$/, '') + '…'
  }
  return lines
}

// Small fixed jitter set so each territory's contour reads as an organic
// blob rather than a perfect ellipse, without needing real path/noise math.
const CONTOUR_JITTER = [1, 0.92, 1.06, 0.96, 1.1, 0.94, 1.03, 1]

function buildContourPath(cx, cy, rx, ry) {
  const points = CONTOUR_JITTER.map((j, i) => {
    const angle = (Math.PI * 2 * i) / CONTOUR_JITTER.length
    return [cx + Math.cos(angle) * rx * j, cy + Math.sin(angle) * ry * j]
  })
  const d = points
    .map(([x, y], i) => {
      if (i === 0) return `M ${x.toFixed(1)} ${y.toFixed(1)}`
      const [px, py] = points[i - 1]
      const mx = (px + x) / 2
      const my = (py + y) / 2
      return `Q ${px.toFixed(1)} ${py.toFixed(1)} ${mx.toFixed(1)} ${my.toFixed(1)}`
    })
    .join(' ')
  return `${d} Z`
}

export default function GraphTerritory({ cluster, faded }) {
  const contour = buildContourPath(cluster.cx, cluster.cy, cluster.rx, cluster.ry)

  return (
    <g className="territory" opacity={faded ? 0.35 : 1} style={{ transition: 'opacity 0.4s ease' }}>
      <ellipse cx={cluster.cx} cy={cluster.cy} rx={cluster.rx} ry={cluster.ry} fill="url(#territoryFade)" />
      <path d={contour} fill="none" stroke="var(--border-soft)" strokeWidth={0.6} opacity={0.35} />
      <text x={cluster.cx} y={cluster.cy - cluster.ry + 22} textAnchor="middle" className="territory-label">
        {cluster.label.toUpperCase()}
      </text>
    </g>
  )
}

  const values = fields
    .map((f) => ({ ...f, value: node.metrics?.[f.key] }))
    .filter((f) => f.value !== undefined && f.value !== null)
    .slice(0, 4)

  // Semantic zoom — a small number of sub-entities revealed only for nodes
  // that define them, rendered as quiet satellites beneath the node rather

        const satY = pos.dy

        // so it reads as a thin tether rather than a spoke through the node body.

import { useMemo, useState, useCallback, useEffect } from 'react'
import { businessNodes, businessEdges } from '../data/businessData'
import { getNeighborhood, getPathEdgeIds } from '../utils/graphUtils'
import { HEALTH, NODE_TYPES } from '../utils/constants'

const CASCADE_STEP_MS = 450 // pacing between sequential cascade reveals

/**
 * Owns all interaction state for the Business Digital Twin graph: hover
 * preview, click-to-focus selection (which now drives "focus mode" —
 * neighborhood highlight + expansion — instead of opening a static
 * inspector card), scenario cascade sequencing, and the compact control
 * toggles (critical-path highlight / risk highlight / signal filter).
 */
export function useBusinessGraph() {
  const [hoveredNodeId, setHoveredNodeId] = useState(null)
  const [selectedNodeId, setSelectedNodeId] = useState(null)
  const [highlightMode, setHighlightMode] = useState(null) // 'dependencies' | 'risk' | null
  const [scenarioHighlight, setScenarioHighlight] = useState(null) // { nodeIds, focusNodeId } | null
  const [signalFilterActive, setSignalFilterActive] = useState(false)
  const [cascadeStep, setCascadeStep] = useState(0) // how many scenario nodes have been "revealed" so far

  const nodes = businessNodes
  const edges = businessEdges

  // Selecting a node now IS focus mode: it establishes the neighborhood
  // used for dimming/expansion, same as hovering did before. Hover still
  // takes priority so a quick preview of another node doesn't require
  // clearing the current focus first.
  const activeNeighborhood = useMemo(() => {
    if (scenarioHighlight) {
      return {
        nodeIds: new Set(scenarioHighlight.nodeIds),
        edgeIds: new Set(
          edges
            .filter((e) => scenarioHighlight.nodeIds.includes(e.source) && scenarioHighlight.nodeIds.includes(e.target))
            .map((e) => e.id)
        ),
      }
    }
    if (hoveredNodeId) return getNeighborhood(edges, hoveredNodeId)
    if (selectedNodeId) return getNeighborhood(edges, selectedNodeId)
    return null
  }, [hoveredNodeId, selectedNodeId, scenarioHighlight, edges])

  // --- Scenario cascade sequencing ---------------------------------------
  // When a scenario becomes active, its affected nodes reveal one at a time
  // rather than all at once, so the graph reads as "the impact travels
  // through the ecosystem" instead of an instant flash.
  useEffect(() => {
    if (!scenarioHighlight) {
      setCascadeStep(0)
      return
    }
    setCascadeStep(1)
    const ids = scenarioHighlight.nodeIds
    const timers = []
    for (let i = 1; i < ids.length; i++) {
      timers.push(setTimeout(() => setCascadeStep(i + 1), i * CASCADE_STEP_MS))
    }
    return () => timers.forEach(clearTimeout)
  }, [scenarioHighlight])

  const cascadeActiveNodeIds = useMemo(() => {
    if (!scenarioHighlight) return new Set()
    return new Set(scenarioHighlight.nodeIds.slice(0, cascadeStep))
  }, [scenarioHighlight, cascadeStep])

  const cascadeActiveEdgeIds = useMemo(() => {
    if (!scenarioHighlight) return new Set()
    return getPathEdgeIds(edges, scenarioHighlight.nodeIds.slice(0, cascadeStep))
  }, [scenarioHighlight, cascadeStep, edges])

  const cascadeInProgress = Boolean(scenarioHighlight) && cascadeStep < (scenarioHighlight?.nodeIds.length ?? 0)

  const riskNodeIds = useMemo(
    () => new Set(nodes.filter((n) => n.health !== HEALTH.HEALTHY).map((n) => n.id)),
    [nodes]
  )

  const criticalEdgeIds = useMemo(() => new Set(edges.filter((e) => e.kind === 'critical').map((e) => e.id)), [edges])
  const criticalNodeIds = useMemo(() => {
    const ids = new Set()
    edges.forEach((e) => {
      if (e.kind === 'critical') {
        ids.add(e.source)
        ids.add(e.target)
      }
    })
    return ids
  }, [edges])

  const signalNodeIds = useMemo(() => new Set(nodes.filter((n) => n.type === NODE_TYPES.SIGNAL).map((n) => n.id)), [nodes])
  const signalEdgeIds = useMemo(() => new Set(edges.filter((e) => e.kind === 'signal').map((e) => e.id)), [edges])

  const isNodeDimmed = useCallback(
    (nodeId) => {
      if (signalFilterActive && signalNodeIds.has(nodeId)) return true
      if (highlightMode === 'risk') return !riskNodeIds.has(nodeId)
      if (highlightMode === 'dependencies') return !criticalNodeIds.has(nodeId)
      if (activeNeighborhood) return !activeNeighborhood.nodeIds.has(nodeId)
      return false
    },
    [activeNeighborhood, highlightMode, riskNodeIds, criticalNodeIds, signalFilterActive, signalNodeIds]
  )

  const isEdgeDimmed = useCallback(
    (edgeId, edge) => {
      if (signalFilterActive && signalEdgeIds.has(edgeId)) return true
      if (highlightMode === 'risk') return !(riskNodeIds.has(edge.source) && riskNodeIds.has(edge.target)) && edge.kind !== 'critical'
      if (highlightMode === 'dependencies') return !criticalEdgeIds.has(edgeId)
      if (activeNeighborhood) return !activeNeighborhood.edgeIds.has(edgeId)
      return false
    },
    [activeNeighborhood, highlightMode, riskNodeIds, criticalEdgeIds, signalFilterActive, signalEdgeIds]
  )

  const isEdgeHighlighted = useCallback(
    (edgeId) => {
      if (scenarioHighlight) return activeNeighborhood?.edgeIds.has(edgeId) ?? false
      if (hoveredNodeId) return activeNeighborhood?.edgeIds.has(edgeId) ?? false
      if (selectedNodeId) return activeNeighborhood?.edgeIds.has(edgeId) ?? false
      return false
    },
    [activeNeighborhood, hoveredNodeId, selectedNodeId, scenarioHighlight]
  )

  const selectedNode = useMemo(() => nodes.find((n) => n.id === selectedNodeId) || null, [nodes, selectedNodeId])

  const selectNode = useCallback((nodeId) => {
    setScenarioHighlight(null)
    setSelectedNodeId((prev) => (prev === nodeId ? null : nodeId))
  }, [])

  const clearSelection = useCallback(() => setSelectedNodeId(null), [])

  const toggleHighlightMode = useCallback((mode) => {
    setHighlightMode((prev) => (prev === mode ? null : mode))
  }, [])

  const resetView = useCallback(() => {
    setSelectedNodeId(null)
    setHighlightMode(null)
    setHoveredNodeId(null)
    setScenarioHighlight(null)
    setSignalFilterActive(false)
  }, [])

  const toggleSignalFilter = useCallback(() => setSignalFilterActive((v) => !v), [])

  const applyScenarioHighlight = useCallback((nodeIds, focusNodeId) => {
    setHighlightMode(null)
    setSelectedNodeId(focusNodeId || null)
    setScenarioHighlight({ nodeIds, focusNodeId })
  }, [])

  const clearScenarioHighlight = useCallback(() => setScenarioHighlight(null), [])

  return {
    nodes,
    edges,
    hoveredNodeId,
    setHoveredNodeId,
    selectedNode,
    selectedNodeId,
    selectNode,
    clearSelection,
    highlightMode,
    toggleHighlightMode,
    isNodeDimmed,
    isEdgeDimmed,
    isEdgeHighlighted,
    resetView,
    applyScenarioHighlight,
    clearScenarioHighlight,
    scenarioHighlight,
    signalFilterActive,
    toggleSignalFilter,
    cascadeActiveNodeIds,
    cascadeActiveEdgeIds,
    cascadeInProgress,
  }
}

import { useEffect, useMemo, useRef, useState } from 'react'

const DEFAULT_RADIUS = 150 // SVG user-space units nodes start reacting within
const MAX_OFFSET = 7 // px a node can be nudged away from the cursor
const MAX_SCALE_BOOST = 0.035

/**
 * Makes the Digital Twin feel "aware" of the user: nodes within `radius` of
 * the cursor drift a few px away and scale up very slightly, falling off
 * smoothly with distance. Disabled entirely when `enabled` is false
 * (prefers-reduced-motion) or while a node is actively selected, so the
 * inspector doesn't fight the graph for stability.
 */
export function useCursorProximity(svgRef, nodes, { enabled = true, radius = DEFAULT_RADIUS } = {}) {
  const [cursor, setCursor] = useState(null) // { x, y } in SVG user space, or null when outside
  const rafRef = useRef(null)
  const pendingPoint = useRef(null)

  useEffect(() => {
    if (!enabled) {
      setCursor(null)
      return
    }
    const svg = svgRef.current
    if (!svg) return

    function toSvgPoint(clientX, clientY) {
      const pt = svg.createSVGPoint()
      pt.x = clientX
      pt.y = clientY
      const ctm = svg.getScreenCTM()
      if (!ctm) return null
      const transformed = pt.matrixTransform(ctm.inverse())
      return { x: transformed.x, y: transformed.y }
    }

    function flush() {
      rafRef.current = null
      if (pendingPoint.current) setCursor(pendingPoint.current)
    }

    function handleMove(e) {
      pendingPoint.current = toSvgPoint(e.clientX, e.clientY)
      if (rafRef.current === null) rafRef.current = requestAnimationFrame(flush)
    }

    function handleLeave() {
      pendingPoint.current = null
      setCursor(null)
    }

    svg.addEventListener('mousemove', handleMove)
    svg.addEventListener('mouseleave', handleLeave)
    return () => {
      svg.removeEventListener('mousemove', handleMove)
      svg.removeEventListener('mouseleave', handleLeave)
      if (rafRef.current) cancelAnimationFrame(rafRef.current)
    }
  }, [svgRef, enabled])

  const offsets = useMemo(() => {
    const map = {}
    if (!cursor) return map

    nodes.forEach((node) => {
      const dx = node.x - cursor.x
      const dy = node.y - cursor.y
      const dist = Math.sqrt(dx * dx + dy * dy)
      if (dist >= radius || dist === 0) return

      const falloff = 1 - dist / radius // 0..1, stronger when closer
      const eased = falloff * falloff // ease-in, calm at the edge of the radius
      const nx = dx / dist
      const ny = dy / dist

      map[node.id] = {
        offsetX: nx * eased * MAX_OFFSET,
        offsetY: ny * eased * MAX_OFFSET,
        scale: 1 + eased * MAX_SCALE_BOOST,
      }
    })

    return map
  }, [cursor, nodes, radius])

  return offsets
}

import { useMemo } from 'react'
import { findNodeById, getNeighborhood } from '../utils/graphUtils'

const MAX_SHIFT = 55 // px — how far the whole graph may drift toward centering the focal node
const SHIFT_LERP = 0.4 // how much of the "full recenter" distance to actually take (partial, not total)
const EXPAND_PUSH = 24 // px a direct neighbor moves away from the focal node
const RECEDE_PUSH = 6 // px an unrelated node drifts further away, reinforcing "focus" without a full re-layout

function clamp(v, min, max) {
  return Math.max(min, Math.min(max, v))
}

/**
 * Computes the visual consequence of "focus mode" for a selected node,
 * without a full force-directed re-layout (which would be unpredictable
 * with a hand-positioned graph). Instead:
 *  - the whole graph nudges partway toward centering the focal node
 *    (clamped so nothing risks clipping outside the viewBox)
 *  - direct neighbors spread outward from the focal node
 *  - everything else recedes very slightly, reinforcing depth
 * The result reads as "the ecosystem reorganized around this entity"
 * while staying deterministic and safe to render at any viewport.
 */
export function useGraphFocus(nodes, edges, focusNodeId, viewW, viewH) {
  return useMemo(() => {
    if (!focusNodeId) return { groupOffset: { x: 0, y: 0 }, nodeOffsets: {}, focalNode: null }

    const focalNode = findNodeById(nodes, focusNodeId)
    if (!focalNode) return { groupOffset: { x: 0, y: 0 }, nodeOffsets: {}, focalNode: null }

    const fullShiftX = viewW / 2 - focalNode.x
    const fullShiftY = viewH / 2 - focalNode.y
    const groupOffset = {
      x: clamp(fullShiftX * SHIFT_LERP, -MAX_SHIFT, MAX_SHIFT),
      y: clamp(fullShiftY * SHIFT_LERP, -MAX_SHIFT, MAX_SHIFT),
    }

    const { nodeIds } = getNeighborhood(edges, focusNodeId)
    const nodeOffsets = {}

    nodes.forEach((n) => {
      if (n.id === focusNodeId) return
      const dx = n.x - focalNode.x
      const dy = n.y - focalNode.y
      const dist = Math.sqrt(dx * dx + dy * dy) || 1
      const nx = dx / dist
      const ny = dy / dist
      const isNeighbor = nodeIds.has(n.id)
      const push = isNeighbor ? EXPAND_PUSH : RECEDE_PUSH // both move outward from the focal node; unrelated nodes just move less
      nodeOffsets[n.id] = { x: nx * push, y: ny * push }
    })

    return { groupOffset, nodeOffsets, focalNode }
  }, [nodes, edges, focusNodeId, viewW, viewH])
}

import { useEffect, useState } from 'react'

/**
 * Tracks the user's prefers-reduced-motion setting so ambient graph motion
 * (drift, cursor proximity, breathing halos) can be disabled outright
 * rather than just slowed down.
 */
export function useReducedMotion() {
  const [reduced, setReduced] = useState(
    () => typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches
  )

  useEffect(() => {
    const mql = window.matchMedia('(prefers-reduced-motion: reduce)')
    const handler = (e) => setReduced(e.matches)
    mql.addEventListener('change', handler)
    return () => mql.removeEventListener('change', handler)
  }, [])

  return reduced
}

// Lightweight graph traversal helpers over the { nodes, edges } business graph shape.
// Kept intentionally simple in Phase 1 — the full dependency traversal engine
// (multi-hop weighting, cascade simulation) arrives in Phase 2.

/** Returns edge ids directly touching a node (either direction). */
export function getDirectEdges(edges, nodeId) {
  return edges.filter((e) => e.source === nodeId || e.target === nodeId)
}

/** Returns node ids immediately upstream (feeding into) a node. */
export function getUpstreamNodeIds(edges, nodeId) {
  return edges.filter((e) => e.target === nodeId).map((e) => e.source)
}

/** Returns node ids immediately downstream (fed by) a node. */
export function getDownstreamNodeIds(edges, nodeId) {
  return edges.filter((e) => e.source === nodeId).map((e) => e.target)
}

/**
 * Returns the set of node ids and edge ids that should be visually
 * highlighted when a given node is hovered/focused: the node itself,
 * plus its immediate upstream and downstream neighbors and connecting edges.
 */
export function getNeighborhood(edges, nodeId) {
  const nodeIds = new Set([nodeId])
  const edgeIds = new Set()

  edges.forEach((e) => {
    if (e.source === nodeId || e.target === nodeId) {
      edgeIds.add(e.id)
      nodeIds.add(e.source)
      nodeIds.add(e.target)
    }
  })

  return { nodeIds, edgeIds }
}

/**
 * Given an ordered list of node ids representing a path (e.g. a scenario's
 * cascade path), returns the set of edge ids connecting consecutive nodes,
 * if such edges exist in the graph.
 */
export function getPathEdgeIds(edges, orderedNodeIds) {
  const edgeIds = new Set()
  for (let i = 0; i < orderedNodeIds.length - 1; i++) {
    const from = orderedNodeIds[i]
    const to = orderedNodeIds[i + 1]
    const match = edges.find(
      (e) => (e.source === from && e.target === to) || (e.source === to && e.target === from)
    )
    if (match) edgeIds.add(match.id)
  }
  return edgeIds
}

export function findNodeById(nodes, id) {
  return nodes.find((n) => n.id === id)
}
